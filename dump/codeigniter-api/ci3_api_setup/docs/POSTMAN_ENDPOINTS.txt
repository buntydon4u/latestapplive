Base URL:
https://your-domain.com/api/v1

Common Headers:
Content-Type: application/json
Authorization: Bearer {{token}}

Response Format:
Success:
{
  "status": true,
  "message": "Data fetched successfully",
  "data": []
}

Error:
{
  "status": false,
  "message": "Invalid credentials"
}

Validation:
{
  "status": false,
  "errors": {
    "username": "Username required"
  }
}

1. Login
POST /login
Headers:
Content-Type: application/json
Body:
{
  "username": "demo",
  "password": "123456"
}
Success Response:
{
  "status": true,
  "message": "Login successful",
  "data": {
    "token": "JWT_TOKEN",
    "parent_selection_required": false,
    "user": {
      "id": 10,
      "name": "Ledger Name",
      "user_type": "ledger",
      "updated_by": "1"
    }
  }
}
Parent Selection Response:
{
  "status": true,
  "message": "Child selection required",
  "data": {
    "token": "PARENT_SELECTION_JWT_TOKEN",
    "parent_selection_required": true,
    "parent_id": 10,
    "name": "Parent Ledger"
  }
}
